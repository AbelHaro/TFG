\section{Objetivos}
\begin{frame}{3. Objetivos}
    \textbf{Objetivo Principal:} Desarrollar un sistema de \textbf{detección de defectos} en objetos en movimiento utilizando CNNs, optimizado para hardware \textbf{NVIDIA}.

        \textbf{Objetivos Específicos:}
        \begin{enumerate}
            \item \textbf{Investigación:} Estado del arte de \textbf{CNNs} y \textbf{aceleradores hardware} (NVIDIA).
            \item \textbf{Datos:} Crear un \textbf{conjunto de datos} robusto.
            \item \textbf{Modelado:} Entrenar y optimizar \textbf{modelos CNN} para la detección en \textbf{tiempo real}.
            \item \textbf{Integración:} \textbf{Sistema de visión artificial} optimizado para hardware NVIDIA Jetson.
            \item \textbf{Optimización:} Identificar \textbf{cuellos de botella} para mejorar el rendimiento.
            \item \textbf{Evaluación:} Evaluar el sistema con métricas clave de \textbf{precisión, latencia y consumo}.
            \item \textbf{Análisis Comparativo:} Determinar la configuración óptima del sistema.
        \end{enumerate}
\end{frame}

\begin{frame}{6. Desarrollo de la solución - Entrenamiento y validación de modelos}

                \item \textcolor{magenta}{\textbf{Captura} de vídeo.}

\begin{frame}{6. Desarrollo de la solución - Segmentación de las etapas (3/3)}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{3. Segmentación por procesos con memoria compartida:}
            \begin{itemize}
                \item Cada etapa se ejecuta en un proceso separado, pero \textbf{comparten memoria}.
                \item $\uparrow$ Permite un procesamiento concurrente con \textbf{menor latencia} en la comunicación.
                \item Las colas de comunicación se implementan sobre \textbf{memoria compartida} entre procesos evitando la sobrecarga de pipes.
            \end{itemize}
        \end{column}

\section{Conclusiones}
\begin{frame}{9. Conclusiones -- Cumplimiento de objetivos (1/2)}
    \textbf{Los objetivos del TFG se han alcanzado con éxito:}
    \begin{itemize}
        \item \textbf{Estudio del estado del arte:} Se realizó un análisis exhaustivo de \textbf{CNNs}, \textbf{aceleradores hardware} y plataformas \textbf{NVIDIA Jetson}, superando desafíos de compatibilidad y configuración en arquitectura ARM64.
        \item \textbf{Creación del conjunto de datos:} Se generó un \textbf{dataset} y vídeos de entrenamiento que simulan condiciones reales de operación con objetos de variabilidad controlada.
        \item \textbf{Entrenamiento de modelos CNN:} Se entrenaron múltiples modelos de las familias \textbf{YOLOv5, YOLOv8 y YOLO11} en diversas precisiones, optimizando hiperparámetros mediante iteraciones exhaustivas.
    \end{itemize}
\end{frame}

\begin{frame}{9. Conclusiones -- Cumplimiento de objetivos (2/2)}
    \begin{itemize}
        \item \textbf{Implementación del sistema integrado:} Se desarrolló un sistema de \textbf{visión artificial} que combina detección con seguimiento \textbf{BYTETrack}, manteniendo la identidad de objetos a lo largo del tiempo.
        \item \textbf{Análisis y optimización de cuellos de botella:} Se exploraron estrategias de segmentación (\textbf{hilos, procesos, memoria compartida, heterogénea}), identificando la \textbf{segmentación por procesos con memoria compartida} como la más efectiva.
        \item \textbf{Evaluación con métricas completas:} Se realizaron experimentos exhaustivos evaluando \textbf{precisión, latencia y consumo} en diferentes configuraciones de hardware Jetson.
    \end{itemize}
\end{frame}

\begin{frame}{6. Desarrollo de la solución - Segmentación de las etapas en Hilos}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{1. Segmentación por hilos:}
            \begin{itemize}
                \item Cada etapa se ejecuta en un hilo separado.
                \item $\uparrow$ La información se comparte entre hilos mediante colas que se consumen de forma asíncrona.
                \item $\downarrow$ Permite un procesamiento paralelo pero no concurrente debido al GIL (Global Interpreter Lock) de Python.
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{center}
                \includegraphics[width=0.75\textwidth]{images/solucion_propuesta/segmentacion_hilos.png}
            \end{center}
        \end{column}
    \end{columns}

\end{frame}

\begin{frame}{6. Desarrollo de la solución - Segmentación de las etapas en Procesos}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{2. Segmentación por procesos:}
            \begin{itemize}
                \item Cada etapa se ejecuta en un \textbf{proceso separado}.
                \item $\uparrow$ Permite un \textbf{procesamiento concurrente}, pero con mayor latencia en la comunicación.
                \item $\downarrow$ La comunicación entre procesos se realiza mediante \textbf{colas} implementadas sobre \textbf{pipes}.
        \end{itemize}
        \end{column}
